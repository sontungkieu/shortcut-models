\section{Checkpoint, đánh giá và các lưu ý vận hành}

\subsection{Checkpoint và logging}

\begin{itemize}
    \item Repo lưu checkpoint qua \filepath{utils/checkpoint.py}.
    \item Trong train loop, code hiện tại nối trực tiếp \code{save_dir} với step number. Vì vậy nên truyền
    \code{save_dir} theo dạng có dấu gạch chéo ở cuối, ví dụ \code{checkpoints/run_a/}.
    \item Chỉ process có chỉ số \(0\) khởi tạo W\&B khi \code{--mode=train}.
\end{itemize}

\subsection{Những điểm nên chú ý trước khi chạy thật}

\begin{enumerate}
    \item Khi load checkpoint, cần truyền lại đúng kiến trúc model đã train, gồm \code{hidden_size},
    \code{patch_size}, \code{depth}, \code{num_heads} và \code{mlp_ratio}.
    \item Với checkpoint \code{sit}, cần truyền lại thêm đúng \code{transport\_path\_type},
    \code{transport\_prediction}, \code{transport\_loss\_weight} và các cờ sampler inference nếu muốn tái lập kết quả.
    \item \code{celebahq256} khác với \code{celeb_a_hq}. Dùng sai tên dataset sẽ fail ngay ở loader.
    \item Inference path phù hợp nhất khi đã chuẩn bị \code{--fid_stats}.
    \item \filepath{helper_inference.py} chỉ lưu \code{x\_render.npy} trong các nhánh mà helper thực sự tích luỹ ảnh render.
    \item \filepath{utils/stable_vae.py} và \filepath{utils/fid.py} đều có bước tải model ngoài ở lần chạy đầu tiên.
\end{enumerate}
